%podatki potrebni za funkcioniranje skripte zunaj izhodišènega programa

Lr=l/1000;
N_plasti=2;
Zq=N*N_plasti;
Kcu=0.38;
d_reza=reza/1000;
b_ds=bds/1000;
h_ds=hs/1000;
D_stroj=Dr/1000+2*d_reza+2*h_ds+2*hys/1000;
Is=In
%tau_u za izraèun širine zoba
t_u=pi()*(Dr/1000+2*d_reza)/Qs;
%širina statorskega utora
b_s=t_u-b_ds;
%dolžina enega ovoja
l_cu=2*Lr+2*t_u+2*b_s;
%specifièna prevodnost bakra
sigma_cu=34e6;
%delovna tempreratura
T_del=80;
%temperaturni koeficient upornosti bakra
a_cu=0.00381;




    mi_selectgroup(3);
    %postavi kolesni kot 90
	mi_moverotate(0,0,4);
    
    iA = In*sqrt(2);
	iB = -iA*0.5;
	iC = iB;
	
	mi_modifycircprop('U',1,iA);
	mi_modifycircprop('V',1,iB);
	mi_modifycircprop('W',1,iC);
	
	mi_analyze();
    mi_loadsolution();

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%doloèi tokovne gostote ter površine utora
mo_groupselectblock(5);
povrsina_utorov = mo_blockintegral(10)/Qs/Lr;
    
fem_gostota_toka=Is*Zq/(Kcu*povrsina_utorov)/1000000;
    
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%doloèitev upornost statorja

Scu_fem=povrsina_utorov/(Zq)*Kcu;

%upornost statorja pri 20st popravljeno Scu na m^2
R_0_fem=2*Zq*l_cu/(sigma_cu*Scu_fem);

%upornost pri delovni temperaturi
R_d_fem=R_0_fem+a_cu*R_0_fem*(T_del-20);

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%Izgube v bakru
p_cu_fem=3*Is^2*R_d_fem

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%del za doloèitev izgub v železu
    b_max_zob_fem=0;
    b_max_jarem_fem=0;
    x_jarem=(0)*cosd(360/(Qs*2))-(Dr/2+reza+hs+hys/2)*sind(360/(Qs*2));
    y_jarem=(0)*sind(360/(Qs*2))+(Dr/2+reza+hs+hys/2)*cosd(360/(Qs*2));

    x_zob=0;
    y_zob=Dr/2+reza+hs/2;
    
    for korak_utora=1:1:Qs
   
    podatki_v_tocki=mo_getpointvalues(x_zob,y_zob);
    
    if sqrt(podatki_v_tocki(2)^2+podatki_v_tocki(3)^2)>b_max_zob_fem
    b_max_zob_fem=sqrt(podatki_v_tocki(2)^2+podatki_v_tocki(3)^2);
    end
    

    podatki_v_tocki=mo_getpointvalues(x_jarem,y_jarem);
    
    if sqrt(podatki_v_tocki(2)^2+podatki_v_tocki(3)^2)>b_max_jarem_fem
    b_max_jarem_fem=sqrt(podatki_v_tocki(2)^2+podatki_v_tocki(3)^2);
    end
    
   x_jarem_1=x_jarem;
   y_jarem_1=y_jarem;
   
   x_zob_1=x_zob;
   y_zob_1=y_zob;
   
    %rotacije toèke
  x_jarem=(x_jarem_1)*cosd(360/(Qs))-(y_jarem_1)*sind(360/(Qs));
  y_jarem=(x_jarem_1)*sind(360/(Qs))+(y_jarem_1)*cosd(360/(Qs));
 
  x_zob=(x_zob_1)*cosd(360/(Qs))-(y_zob_1)*sind(360/(Qs));
  y_zob=(x_zob_1)*sind(360/(Qs))+(y_zob_1)*cosd(360/(Qs));
    
    end

    
   b_max_jarem_fem
   b_max_zob_fem
